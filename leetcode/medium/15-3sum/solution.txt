ms[lo], nums[hi]])

                    lo += 1
                    hi -= 1

                    # Skip duplicate left values
                    while lo < hi and nums[lo] == nums[lo - 1]:
                        lo += 1

                    # Skip duplicate right values
                    while lo < hi and nums[hi] == nums[hi + 1]:
                        hi -= 1

                elif summ < 0:
                    lo += 1

                else:
                    hi -= 1
