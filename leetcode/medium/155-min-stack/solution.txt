       elif self.min_stack[-1] < value:
            self.min_stack.append(self.min_stack[-1])
        
        else:
            self.min_stack.append(value)
        

        

    def pop(self):
        """
        :rtype: None
        """
        self.stack.pop()
        self.min_stack.pop()

    def top(self):
        """
        :rtype: int
        """
        return self.stack[-1]

        

    def getMin(self):
        """
        :rtype: int
        """
        return self.min_stack[-1]
